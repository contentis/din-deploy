cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(din_audio_workbench LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    include(CTest)
    set(DIN_AUDIO_WITH_ASR OFF)
else()
    set(DIN_AUDIO_WITH_ASR ON)
endif()
if(NOT TARGET din_ui)
    add_subdirectory(../ui ${CMAKE_CURRENT_BINARY_DIR}/ui)
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/miniaudio.cmake)
if(NOT TARGET nlohmann_json::nlohmann_json)
    include(../../cmake/nlohmann_json.cmake)
endif()
set(audio_sources main.cpp workbench.cpp settings.cpp audio_device.cpp transcript.cpp checks.cpp inference.cpp preferences.cpp)
if(DIN_AUDIO_WITH_ASR)
    add_din_executable(din_audio_ui ${audio_sources})
    target_compile_definitions(din_audio_ui PRIVATE DIN_AUDIO_WITH_ASR=1)
    target_link_libraries(din_audio_ui PRIVATE din_asr_qwen3 din_asr_whisper din_asr_parakeet_tdt din_asr_nemotron din_diarization)
else()
    add_executable(din_audio_ui ${audio_sources})
endif()
target_include_directories(din_audio_ui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../common)
if(MSVC)
    target_compile_options(din_audio_ui PRIVATE /utf-8 /W4)
    target_compile_definitions(din_audio_ui PRIVATE NOMINMAX)
endif()
target_link_libraries(din_audio_ui PRIVATE din_ui din_audio nlohmann_json::nlohmann_json)
if(BUILD_TESTING)
    add_test(NAME audio_ui_worker_contract COMMAND din_audio_ui --self-test)
endif()

# Optional native integration test for a machine with exported models / RTX.
set(DIN_AUDIO_TEST_WHISPER_MODEL "" CACHE PATH "Whisper export for audio UI integration test")
set(DIN_AUDIO_TEST_ALIGNER_MODEL "" CACHE PATH "Qwen aligner export for audio UI integration test")
set(DIN_AUDIO_TEST_RECORDING "" CACHE FILEPATH "Long recording for audio UI integration test")
if(BUILD_TESTING AND DIN_AUDIO_WITH_ASR AND DIN_AUDIO_TEST_WHISPER_MODEL AND
   DIN_AUDIO_TEST_ALIGNER_MODEL AND DIN_AUDIO_TEST_RECORDING)
    add_test(NAME audio_ui_native_alignment COMMAND din_audio_ui --alignment-check 1
        "${DIN_AUDIO_TEST_WHISPER_MODEL}" "${DIN_AUDIO_TEST_RECORDING}" "${DIN_AUDIO_TEST_ALIGNER_MODEL}" trt-rtx)
endif()
